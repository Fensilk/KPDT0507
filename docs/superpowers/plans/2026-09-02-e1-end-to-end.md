```python
"""共享 LoRA：低秩适配注入 + 合并。autodl 无需 peft。"""
import torch
import torch.nn as nn

class LoRALinear(nn.Module):
    def __init__(self, original: nn.Linear, r: int = 16, alpha: int = 16):
        super().__init__()
        self.original = original
        self.original.weight.requires_grad_(False)
        if self.original.bias is not None:
            self.original.bias.requires_grad_(False)
        self.scaling = alpha / r
        self.lora_A = nn.Parameter(torch.zeros(r, original.in_features))
        self.lora_B = nn.Parameter(torch.zeros(original.out_features, r))
        nn.init.kaiming_uniform_(self.lora_A, a=5 ** 0.5)
        nn.init.zeros_(self.lora_B)

    def forward(self, x):
        return self.original(x) + self.scaling * (x @ self.lora_A.t() @ self.lora_B.t())

def inject_lora(model, r=16, alpha=16, targets=("query", "key", "value")):
    for layer in model.encoder.layer:
        attn = layer.attention.attention
        for name in targets:
            setattr(attn, name, LoRALinear(getattr(attn, name), r, alpha))
        layer.attention.output.dense = LoRALinear(layer.attention.output.dense, r, alpha)

def merge_lora(model):
    """LoRALinear 的 A/B 合并回权重，替换为普通 Linear（推理用）。"""
    def _merge(module, name):
        mod = getattr(module, name)
        if isinstance(mod, nn.Linear):
            return
        delta = (mod.scaling * (mod.lora_B @ mod.lora_A)).to(mod.original.weight.dtype)
        lin = nn.Linear(mod.original.in_features, mod.original.out_features,
                        bias=mod.original.bias is not None)
        lin.weight.data = mod.original.weight.data + delta
        if mod.original.bias is not None:
            lin.bias.data = mod.original.bias.data
        setattr(module, name, lin)

    for layer in model.encoder.layer:
        attn = layer.attention.attention
        for name in ("query", "key", "value"):
            _merge(attn, name)
        _merge(layer.attention.output, "dense")
```

```python
"""E1 连续帧缓存：与 E3 同批 2,000 训练视频，抽全部 80 连续帧。"""
import os, sys, argparse, cv2, numpy as np, pandas as pd, torch
BASE = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, os.path.join(BASE, "preprocessing"))
from av_decode import read_rgb_frames

NPZ = os.path.join(BASE, "data", "omnifall_dinov2_giant_frame.npz")
VIDEO_ROOT = os.path.join(BASE, "DATASET-omnifall", "data_files", "extracted")
TRAIN_CSV = os.path.join(BASE, "DATASET-omnifall", "splits", "syn", "random", "train.csv")
OUT = os.path.join(BASE, "data", "p9e1_frames")
SIZE = (224, 224)

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--n_videos", type=int, default=2000)
    ap.add_argument("--seed", type=int, default=42)
    args = ap.parse_args()

    d = np.load(NPZ, allow_pickle=True, mmap_mode="r")
    vp = [str(p) for p in d["video_paths"]]
    path2idx = {p: i for i, p in enumerate(vp)}

    train_paths = list(pd.read_csv(TRAIN_CSV)["path"].str.strip())
    rng = np.random.default_rng(args.seed)   # 与 E3 同 seed → 同批视频
    rng.shuffle(train_paths)
    selected = train_paths[: args.n_videos]

    os.makedirs(OUT, exist_ok=True)
    n_ok = 0
    for i, rel in enumerate(selected):
        mp4 = os.path.join(VIDEO_ROOT, rel + ".mp4")
        if rel not in path2idx or not os.path.exists(mp4):
            continue
        frames = read_rgb_frames(mp4, 80)      # PyAV, list of (H,W,3) uint8
        imgs = np.stack([cv2.resize(f, SIZE, interpolation=cv2.INTER_LINEAR)
                         for f in frames]).astype(np.uint8)  # (80,224,224,3)
        torch.save({"frames": imgs, "video": rel},
                   os.path.join(OUT, f"{n_ok:05d}_{rel.replace('/', '_')}.pt"))
        n_ok += 1
        if (i + 1) % 500 == 0:
            print(f"  [{i+1}/{len(selected)}] 已存 {n_ok}")
    print(f"完成：{n_ok} 视频 → {OUT}")
    # 冒烟验证
    f0 = torch.load(os.path.join(OUT, "00000_lie_down_lie_down_to_088.pt") if n_ok else None)
    if f0: print("首视频帧均值:", round(float(f0["frames"].mean()), 1))  # >20 为真实帧

if __name__ == "__main__":
    main()
```

```python
"""E1 端到端模型：ViT-g(LoRA) 逐帧 CLS → Δdiff(3072d) → Phase6 时序头。"""
import os, sys
import torch, torch.nn as nn
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from experiments.lora import inject_lora          # noqa: E402
from models.phase6_model import Phase6TernaryModel  # noqa: E402

CLS_DIM = 1536  # ViT-g

class E1Model(nn.Module):
    """frames (B,T,3,H,W) → logits (B,T,3)。vit 主干 fp16，头 fp32(autocast)。"""
    def __init__(self, vit_model, temporal=None, lora_rank=16):
        super().__init__()
        self.vit = vit_model            # facebook/dinov2-giant（已 .half() 冻结主干）
        self.temporal = temporal or Phase6TernaryModel(
            input_dim=CLS_DIM * 2, hidden_dim=384, decoder_type="transformer",
            causal=False, num_heads=6, dropout=0.3, max_len=100, num_classes=3)
        self.temporal.input_proj.to(torch.float32)

    def forward(self, frames):
        B, T = frames.shape[0], frames.shape[1]
        x = frames.reshape(B * T, *frames.shape[2:])          # (B*T,3,H,W)
        with torch.cuda.amp.autocast(dtype=torch.float16):
            cls = self.vit(x).last_hidden_state[:, 0, :]      # (B*T,1536) fp16
        cls = cls.reshape(B, T, CLS_DIM).float()              # (B,T,1536)
        diff = torch.zeros_like(cls)
        diff[:, 1:] = cls[:, 1:] - cls[:, :-1]                # Δdiff, diff[0]=0
        feats = torch.cat([cls, diff], dim=-1)                # (B,T,3072)
        return self.temporal(feats)                           # (B,T,3)

    @classmethod
    def from_pretrained(cls, model_name="facebook/dinov2-giant", lora_rank=16,
                        device="cuda"):
        from transformers import AutoModel
        vit = AutoModel.from_pretrained(model_name).half()
        for p in vit.parameters():
            p.requires_grad_(False)
        inject_lora(vit, r=lora_rank)
        return cls(vit, lora_rank=lora_rank).to(device)
```

```python
"""E1 训练：窗口采样 + 端到端反传 + val 定时检查 + 发散停。"""
import os, sys, glob, json, time, argparse
import numpy as np, pandas as pd, torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader
BASE = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, BASE)
sys.path.insert(0, os.path.join(BASE, "preprocessing"))
from models.phase9_e1_model import E1Model

def ternary(l16):
    return np.where(l16 == 1, 0, np.where(l16 == 2, 1, 2))

class WindowCache(Dataset):
    """每视频所有 stride 窗口。frames from .pt (80,224,224,3), labels from NPZ。"""
    def __init__(self, cache_dir, npz, window=64, stride=16):
        self.files = sorted(glob.glob(os.path.join(cache_dir, "*.pt")))
        d = np.load(npz, allow_pickle=True, mmap_mode="r")
        self.labels16 = d["labels_16"]
        self.vsi = d["video_start_indices"]
        vp = [str(p) for p in d["video_paths"]]
        self.idx = {p: i for i, p in enumerate(vp)}
        self.window, self.stride = window, stride
        self.meta = []  # (file, video_rel, st, win_start)
        for f in self.files:
            video = torch.load(f, weights_only=False)["video"]
            vi = self.idx[video]; st = int(self.vsi[vi])
            n_frames = 80
            for ws in range(0, n_frames - window + 1, stride):
                self.meta.append((f, video, st, ws))

    def __len__(self):
        return len(self.meta)

    def __getitem__(self, i):
        f, video, st, ws = self.meta[i]
        frames = torch.load(f, weights_only=False)["frames"][ws:ws + self.window]
        y = ternary(self.labels16[st + ws: st + ws + self.window])
        return torch.from_numpy(np.ascontiguousarray(frames)).float(), torch.from_numpy(y).long()
```

- [ ] **Step 2: 监控函数 + 训练主循环**

```python
def val_check(model, val_paths_cache, npz, window, device, n_videos=150):
    """在 val 视频子集跑端到端，返回 (avg_f1, fallen_prec, fallen_rec, loss)。"""
    model.eval()
    # 复用 WindowCache(val), 只取前 n 个不同视频的窗口
    ds = WindowCache(val_paths_cache, npz, window, 80)   # stride=80 → 每视频1窗口
    all_y, all_p = [], []
    for i in range(min(len(ds), n_videos)):
        fr, y = ds[i]
        with torch.no_grad():
            logits = model(fr.unsqueeze(0).to(device))
        all_p.append(logits.argmax(-1).cpu().numpy().ravel())
        all_y.append(y.numpy())
    y = np.concatenate(all_y); p = np.concatenate(all_p)
    from sklearn.metrics import f1_score, precision_score, recall_score
    def pr(cls):
        return (precision_score(y, p, labels=[0,1,2], average=None, zero_division=0)[cls],
                recall_score(y, p, labels=[0,1,2], average=None, zero_division=0)[cls])
    fp, fr_ = pr(1)
    avg = f1_score(y, p, labels=[0,1,2], average=None, zero_division=0).mean()
    model.train()
    return {"val_avg_f1": float(avg), "val_fallen_prec": float(fp), "val_fallen_rec": float(fr_)}
```

主循环伪码（完整见实现）：窗口 shuffle → batch=2 → autocast 前向 → CE(labels) 反传 → 每 batch 记 loss + 分 class acc；`step % val_every == 0` 跑 `val_check`；连续 2 次 val_avg_f1 下滑 → early stop；每 epoch 存 checkpoint。
